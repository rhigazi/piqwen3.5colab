             ⚠️ "--no-binary llama-cpp-python" ist PFLICHT — sonst nimmt

               --n_gpu_layers -1 --n_ctx 8192 --host 0.0.0.0 --port 8000
  Schritt 5  Health-Check (max. 300s) + /v1/models + VRAM

  Schritt 8  Abnahme: nvidia-smi (VRAM-Belegung)

    ['python3', '-m', 'llama_cpp.server',
     '--model', '/content/models/qwen3.5-4B-super-coder.Q4_0.gguf',
     '--n_gpu_layers', '-1',
     '--n_ctx', '8192',
     '--host', '0.0.0.0', '--port', '8000'],
    stdout=open('/content/logs/llama_server.log', 'w'),
    stderr=subprocess.STDOUT,
    start_new_session=True,
)

ZELLE 6 — Inference-Test (OpenAI-kompatible API)
from openai import OpenAI
import time
client = OpenAI(base_url='http://localhost:8000/v1', api_key='no-key-required')
t0 = time.time()
resp = client.chat.completions.create(
    model='qwen3.5-4B-super-coder.Q4_0',
    messages=[
        {'role': 'system', 'content': 'You are a helpful coding assistant.'},
        {'role': 'user', 'content': 'Schreibe eine Python-Funktion fuer Binary Search. Nur Code.'},
    ],
    temperature=0.6, top_p=0.95, max_tokens=512,
)
dt = time.time() - t0
print(f'[{dt:.1f}s | {resp.usage.completion_tokens/dt:.0f} tok/s]')
print(resp.choices[0].message.content)

ZELLE 8 — Pi-Provider konfigurieren (frische VM)
import json, os
cfg = {
    'providers': {
        'llama-cpp': {
            'baseUrl': 'http://localhost:8000/v1',
            'api': 'openai-completions',
            'apiKey': 'no-key-required',
            'compat': {
                'supportsDeveloperRole': False,
                'supportsReasoningEffort': False,
            },
            'models': [{
                'id': 'qwen3.5-4B-super-coder.Q4_0',
                'name': 'Qwen 3.5 4B Super Coder (local Q4_0)',
                'reasoning': True,
                'input': ['text'],
                'contextWindow': 32768,
                'maxTokens': 8192,
                'cost': {'input': 0, 'output': 0, 'cacheRead': 0, 'cacheWrite': 0},
            }],
        }
    }
}
os.makedirs(os.path.expanduser('~/.pi/agent'), exist_ok=True)
with open(os.path.expanduser('~/.pi/agent/models.json'), 'w') as f:
    json.dump(cfg, f, indent=2)
